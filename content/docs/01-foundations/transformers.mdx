## Introduction

The Transformer architecture revolutionized deep learning. This guide implements multi-head attention, positional encoding, and a causal language model from scratch using PyTorch.

## Scaled Dot-Product Attention

The core operation:

`Attention(Q, K, V) = softmax(Q * K^T / sqrt(d_k)) * V`

```python
import torch
import torch.nn.functional as F

def scaled_dot_product_attention(
    Q: torch.Tensor, K: torch.Tensor, V: torch.Tensor, mask: torch.Tensor | None = None
) -> torch.Tensor:
    scores = Q @ K.transpose(-2, -1) / (K.size(-1) ** 0.5)
    if mask is not None:
        scores = scores.masked_fill(mask == 0, float("-inf"))
    return F.softmax(scores, dim=-1) @ V
```

## Multi-Head Attention

```python
class MultiHeadAttention(torch.nn.Module):
    def __init__(self, d_model: int, n_heads: int):
        super().__init__()
        self.W_q = torch.nn.Linear(d_model, d_model)
        self.W_k = torch.nn.Linear(d_model, d_model)
        self.W_v = torch.nn.Linear(d_model, d_model)
        self.W_o = torch.nn.Linear(d_model, d_model)
        self.n_heads = n_heads

    def forward(self, x: torch.Tensor, mask: torch.Tensor | None = None) -> torch.Tensor:
        B, T, D = x.shape
        Q = self.W_q(x).view(B, T, self.n_heads, D // self.n_heads).transpose(1, 2)
        K = self.W_k(x).view(B, T, self.n_heads, D // self.n_heads).transpose(1, 2)
        V = self.W_v(x).view(B, T, self.n_heads, D // self.n_heads).transpose(1, 2)
        out = scaled_dot_product_attention(Q, K, V, mask)
        out = out.transpose(1, 2).contiguous().view(B, T, D)
        return self.W_o(out)
```

## Causal Language Model Head

```python
class GPT(torch.nn.Module):
    def __init__(self, vocab_size: int, d_model: int, n_layers: int, n_heads: int):
        super().__init__()
        self.token_embedding = torch.nn.Embedding(vocab_size, d_model)
        self.pos_embedding = torch.nn.Embedding(1024, d_model)
        self.layers = torch.nn.ModuleList([
            torch.nn.TransformerEncoderLayer(d_model, n_heads, dim_feedforward=d_model * 4)
            for _ in range(n_layers)
        ])
        self.lm_head = torch.nn.Linear(d_model, vocab_size)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        B, T = x.shape
        pos = torch.arange(0, T, device=x.device).unsqueeze(0)
        x = self.token_embedding(x) + self.pos_embedding(pos)
        causal_mask = torch.triu(torch.ones(T, T, device=x.device), diagonal=1).bool()
        for layer in self.layers:
            x = layer(x, src_key_padding_mask=None, mask=causal_mask)
        return self.lm_head(x)
```

## Key Takeaways

- Attention computes weighted combinations of all positions
- Multi-head attention allows the model to attend to different representation subspaces
- Causal masking prevents the model from seeing future tokens
- Positional encoding gives the model sequence order information
